title AmanPay - Diagramme de Classes

class Utilisateur {
  id: int
  nom: string
  email: string
  telephone: string
  motDePasse: string
  role: string
}

class Wallet {
  id: int
  soldeTotal: float
  devise: string
}

class Transfert {
  id: int
  montantTotal: float
  date: Date
  type: string
  statut: string
}

class Enveloppe {
  id: int
  type: string
  montant: float
  statut: string
}

class Transaction {
  id: int
  montant: float
  type: string
  date: Date
  statut: string
}

class Partenaire {
  id: int
  nom: string
  type: string
  localisation: string
}

class Facture {
  id: int
  reference: string
  montant: float
  date: Date
  statut: string
}

class DemandeDeblocage {
  id: int
  motif: string
  statut: string
  date: Date
}

class DemandeArgent {
  id: int
  montant: float
  categorie: string
  statut: string
  date: Date
}

class Historique {
  id: int
  action: string
  date: Date
}

' Relations
Utilisateur "1" -- "1" Wallet
Utilisateur "1" -- "*" Transfert : envoie
Utilisateur "1" -- "*" Transfert : reçoit

Transfert "1" -- "*" Enveloppe
Enveloppe "1" -- "*" Transaction

Transaction "*" -- "1" Partenaire
Transaction "1" -- "0..1" Facture

Enveloppe "1" -- "*" DemandeDeblocage
Utilisateur "1" -- "*" DemandeDeblocage

Utilisateur "1" -- "*" DemandeArgent
Utilisateur "1" -- "*" Historique
